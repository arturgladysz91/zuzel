    public static RobustEpisode Execute(string d,string raw,RobustBudget b,string category,string name,RiderDecisionCandidate c,FreeLearningScenario s,bool two,double? interveneAt=null,FreeRacingAction? alternative=null){
        if(b.Entries.Sum(e=>e.Forecasts)>1480000)throw new InvalidOperationException("Forecast reserve exhausted.");
        var run=RobustLearningExperiment.Execute(d,raw,b,category,name,c.Id,c.Policy,s,twoBends:two,horizonSeconds:c.Horizon,normalizeProgress:true,
            fullThrottleCandidate:c.FullThrottle,decisionObjective:c.NewObjective,interveneAt:interveneAt,intervention:alternative);
        if(run.Metrics.Native.Outcome=="projection")throw new InvalidOperationException("Projection model disagreement; stop experiment.");

    }
    private static double Median(IEnumerable<double> values){var a=values.Order().ToArray();return a.Length%2==1?a[a.Length/2]:(a[a.Length/2-1]+a[a.Length/2])/2;}
    public static RiderDecisionTimeScore TimeScore(RobustEvaluation e,RobustEvaluation baseline){
        if(!e.Scenarios.SequenceEqual(baseline.Scenarios))throw new ArgumentException("Paired selection states must match.");
        var all=e.Results.Zip(baseline.Results,(a,b)=>(a,b)).Select((p,i)=>(p.a,p.b,Two:e.Scenarios[i].EndsWith("-two",StringComparison.Ordinal)))
            .Where(p=>p.a.Native.Completed&&p.b.Native.Completed).Select(p=>(p.Two,Ratio:p.a.Native.GateSeconds!.Value/p.b.Native.GateSeconds!.Value-1)).ToArray();
        var two=all.Where(p=>p.Two).Select(p=>p.Ratio).ToArray();var single=all.Where(p=>!p.Two).Select(p=>p.Ratio).ToArray();
        double? worst=all.Length>0?all.Max(p=>p.Ratio):null;return new(all.Length,two.Length>0?Median(two):null,single.Length>0?Median(single):null,worst,
            worst is not null&&worst<=.15,worst is{} w?Math.Max(0,w-.15):null);
    }
    private static int Severity(RobustEvaluation e)=>e.Results.Sum(r=>r.Native.Completed?0:r.Native.Outcome switch{"confirmed crossing" or "control envelope"=>3,"reverse"=>2,"uncertified interval"=>1,_=>4});
    public static IOrderedEnumerable<RobustEvaluation> RankR1(IEnumerable<RobustEvaluation> scores,RobustEvaluation baseline)=>scores
        .OrderByDescending(RobustRanking.Completed).ThenBy(Severity).ThenByDescending(FullThrottleLearningExperiment.TwoCompleted)
        .ThenByDescending(e=>TimeScore(e,baseline).WithinWorstBound).ThenBy(e=>TimeScore(e,baseline).WorstExcess??double.PositiveInfinity)
        .ThenBy(e=>TimeScore(e,baseline).MedianTwo??double.PositiveInfinity).ThenBy(e=>TimeScore(e,baseline).MedianSingle??double.PositiveInfinity)
        .ThenBy(e=>TimeScore(e,baseline).WorstRatio??double.PositiveInfinity).ThenByDescending(e=>e.Results.Min(r=>r.Native.MinimumClearanceMeters))
        .ThenByDescending(e=>e.Results.Where(r=>r.Native.Completed).Select(r=>r.Native.ExitSpeed!.Value).DefaultIfEmpty(0).Min())
        .ThenBy(e=>e.Results.Where(r=>r.Native.Completed).Select(r=>Math.Abs(r.ExitYawRate??0)).DefaultIfEmpty(double.PositiveInfinity).Max()).ThenBy(e=>e.Id);

    private static void Train(string d,string raw,RobustBudget b,string label,uint seed,bool newer){
        var file="generations-"+label+".json";var gs=File.Exists(Path.Combine(d,file))?Read<List<RobustGeneration>>(d,file):new();
        var random=new FreeRacingExperiment.RandomStream(seed);for(int i=0;i<gs.Count*32;i++)random.Normal();
        var incumbent=gs.Count>0?gs[^1].BestPolicy:StartPolicy();var mean=gs.Count>0?gs[^1].Mean:incumbent.Values;var std=gs.Count>0?gs[^1].Std:mean.Select((_,i)=>(FreeLearningPolicy.Upper[i]-FreeLearningPolicy.Lower[i])*.3).ToArray();
        for(int g=gs.Count;g<5;g++){
            if(b.Entries.Sum(e=>e.CpuSeconds)>5000||b.Entries.Sum(e=>e.Evaluations)>90000000||b.Entries.Sum(e=>e.Forecasts)>1000000)throw new InvalidOperationException("Independent evaluation reserve reached.");
            var cpu=Process.GetCurrentProcess().TotalProcessorTime;var cases=Schedule(g);var cs=new List<RobustEvaluation>();
            for(int c=0;c<5;c++){
                var id=10000+(seed==Seeds[1]?2000:0)+(newer?1000:0)+g*5+c;var policy=RobustLearningExperiment.DrawPolicy(random,incumbent,mean,std,c);var candidate=new RiderDecisionCandidate(label,id,policy,newer);
                cs.Add(new(label,g,id,policy,cases.Select((s,i)=>s.Name+(i>0?"-two":"-single")).ToArray(),cases.Select((s,i)=>Execute(d,raw,b,"train-ab",$"{label}-g{g:D2}-p{id}-"+s.Name,candidate,s,i>0).Metrics).ToArray()));
            }
            var ranked=FullThrottleLearningExperiment.Rank(cs).ToArray();incumbent=ranked[0].Policy;
            for(int i=0;i<8;i++){var m=ranked.Take(2).Average(c=>c.Policy.Values[i]);mean[i]=.4*mean[i]+.6*m;std[i]=Math.Max(.15*(FreeLearningPolicy.Upper[i]-FreeLearningPolicy.Lower[i]),Math.Sqrt(ranked.Take(2).Average(c=>Math.Pow(c.Policy.Values[i]-m,2))));}
            var generation=new RobustGeneration(label,g,(g+1)*20,cs.ToArray(),ranked[0].Id,incumbent,(double[])mean.Clone(),(double[])std.Clone(),(Process.GetCurrentProcess().TotalProcessorTime-cpu).TotalSeconds,
                cs.Sum(c=>c.Results.Sum(r=>r.Native.PhysicsEvaluations)),cs.Sum(c=>c.Results.Sum(r=>r.Native.Forecasts)));gs.Add(generation);Save(d,file,gs);
            Console.WriteLine($"GEN {label}/{g}: {cs.Sum(RobustRanking.Completed)}/20 two={cs.Sum(FullThrottleLearningExperiment.TwoCompleted)}/15 best={ranked[0].Id} CPU={generation.CpuSeconds:F2}");
            // User-authorized scheduling amendment after the first complete generation.
            // All aggregate budgets and all scientific choices remain frozen.
            var generationCpuLimit=File.Exists(Path.Combine(d,"budget-amendment.json"))?600:225;
            if(generation.CpuSeconds>generationCpuLimit)throw new InvalidOperationException("Generation CPU limit.");
        }
    }
    private static RiderDecisionCandidate[] Shortlist(string d,string label,bool newer){
        var gs=Read<RobustGeneration[]>(d,"generations-"+label+".json");var even=gs.Where(g=>g.Generation%2==0).SelectMany(g=>g.Candidates).ToArray();
        var speed=even.OrderByDescending(FullThrottleLearningExperiment.TwoCompleted).ThenByDescending(RobustRanking.Completed).ThenBy(Severity)
            .ThenBy(e=>FullThrottleLearningExperiment.Two(e).All(r=>r.Native.Completed)?Median(FullThrottleLearningExperiment.Two(e).Select(r=>r.Native.GateSeconds!.Value)):double.PositiveInfinity)
            .ThenBy(RobustRanking.WorstTime).ThenByDescending(e=>e.Results.Min(r=>r.Native.MinimumClearanceMeters)).ThenBy(e=>e.Id).First();
        var last=FullThrottleLearningExperiment.Rank(gs[^1].Candidates).First();var anchor=FullThrottleLearningExperiment.Rank(gs.Where(g=>g.Generation%2==0).Select(g=>FullThrottleLearningExperiment.Rank(g.Candidates).First())).First();
        return new[]{last,anchor,speed}.DistinctBy(x=>Hash(x.Policy)).Select(x=>new RiderDecisionCandidate(label,x.Id,x.Policy,newer)).ToArray();
    }
    private static void Select(string d,string raw,RobustBudget b){
        if(!File.Exists(Path.Combine(d,"training-complete.json"))||b.Entries.Any(e=>e.Category=="final"))throw new InvalidOperationException("Invalid selection stage.");
        var cs=Seeds.SelectMany(seed=>new[]{false,true}.SelectMany(newer=>Shortlist(d,$"{(newer?"B":"A")}-s{seed}",newer))).Concat(new[]{new RiderDecisionCandidate("baseline4061-full",4061,StartPolicy(),false),
            new RiderDecisionCandidate("reference132",6022,Read<ThrottleCandidate>(PhysicalSteeringAudit.PreviousPath("../full-throttle-learning"),"winner.json").Policy,false)}).ToArray();Save(d,"selection-candidates.json",cs);
        var scores=new List<RobustEvaluation>();foreach(var c in cs){var results=Selection.Select((s,i)=>Execute(d,raw,b,"selection",$"select-{c.Label}-{c.Id}-"+s.Name,c,s,i>=2).Metrics).ToArray();
            scores.Add(new(c.Label,-1,c.Id,c.Policy,Selection.Select((s,i)=>s.Name+(i>=2?"-two":"-single")).ToArray(),results));Save(d,"selection.json",scores);}
        var baseline=scores.Single(s=>s.Variant=="baseline4061-full");Save(d,"selection-time-scores.json",scores.Select(s=>new{s.Variant,s.Id,Score=TimeScore(s,baseline)}));
        var choices=new List<RiderDecisionCandidate>();foreach(var newer in new[]{false,true})foreach(var r in new[]{0,1}){
            var pool=scores.Where(s=>s.Variant.StartsWith(newer?"B-":"A-",StringComparison.Ordinal));var selected=(r==0?FullThrottleLearningExperiment.Rank(pool):RankR1(pool,baseline)).First();
            var candidate=cs.Single(c=>c.Label==selected.Variant&&c.Id==selected.Id);choices.Add(candidate with{Label=(newer?"B":"A")+r});
        }
        Save(d,"ranked-choices.json",choices);
    }
    private static void Freeze(string d,RobustBudget b){
        if(File.Exists(Path.Combine(d,"winner-lock.json"))||b.Entries.Any(e=>e.Category=="final"))throw new InvalidOperationException("Choice already sealed/final opened.");
        var choices=Read<RiderDecisionCandidate[]>(d,"ranked-choices.json");var main=choices.Single(c=>c.Label=="B1");Save(d,"winner.json",main);
        var plan=new[]{new RiderDecisionCandidate("130-H06",1020,ExtendedLearningExperiment.StartPolicy(),false,false,.6),new RiderDecisionCandidate("131-H09",4061,StartPolicy(),false,false),
            new RiderDecisionCandidate("132-H09",6022,Read<ThrottleCandidate>(PhysicalSteeringAudit.PreviousPath("../full-throttle-learning"),"winner.json").Policy,false),main with{Label="133-B1"}}
            .Concat(choices.Where(c=>c.Label!="B1").Select(c=>c with{Label="133-"+c.Label})).ToArray();
        Save(d,"final-plan.json",plan);Save(d,"winner-lock.json",new{PolicyHash=Hash(main),PlanHash=Hash(plan),FinalWorlds=0,ReservationCount=b.Entries.Count,Frozen=true});
    }
    private static string Semantics(RiderDecisionCandidate c)=>Hash(new{c.Policy,c.NewObjective,c.FullThrottle,c.Horizon});
    private static void Final(string d,string raw,RobustBudget b){
        var winner=Read<RiderDecisionCandidate>(d,"winner.json");var plan=Read<RiderDecisionCandidate[]>(d,"final-plan.json");var seal=Read<JsonElement>(d,"winner-lock.json");
        if(Hash(winner)!=seal.GetProperty("PolicyHash").GetString()||Hash(plan)!=seal.GetProperty("PlanHash").GetString())throw new InvalidOperationException("Frozen choice changed.");
        var aliases=plan.ToDictionary(c=>c.Label,c=>plan.First(p=>Semantics(p)==Semantics(c)).Label);Save(d,"final-aliases.json",aliases);var evaluations=new List<RobustEvaluation>();
        foreach(var c in plan)foreach(var two in new[]{false,true}){
            var suffix=two?"-two":"-single";var cases=two?FinalTwo:FinalSingle;RobustMetrics[] results;
            if(aliases[c.Label]!=c.Label)results=evaluations.Single(e=>e.Variant==aliases[c.Label]+suffix).Results;
            else results=cases.Select(s=>Execute(d,raw,b,"final","final-"+c.Label+"-"+s.Name,c,s,two).Metrics).ToArray();
            evaluations.Add(new(c.Label+suffix,-1,c.Id,c.Policy,cases.Select(s=>s.Name).ToArray(),results));Save(d,"final-results.json",evaluations);
        }
    }
